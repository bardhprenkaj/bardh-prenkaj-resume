{{/* Leaflet map of the `places` list in the current page's front matter. Each place needs: name, location, lat, lng. Optional: dates, people (Markdown), people_label (default "Worked with"), logo (file name in assets/media/icons/brands/, without .svg; replaces the card title), card_text (Markdown; shown only in the detail card). Hovering a point opens its popup; clicking keeps it open. "Check details" in a popup shrinks the map to the left, centers it on the place and slides in a card on the right. Usage: {{< places_map >}} */}} {{ $places := .Page.Params.places }} {{ $id := printf "places-map-%d" .Ordinal }} {{ $points := slice }} {{ range $k, $place := $places }} {{ $info := printf "
%s
" (.location | htmlEscape) }} {{ with .dates }}{{ $info = printf "%s
%s
" $info (. | htmlEscape) }}{{ end }} {{ with .people }} {{ $info = printf "%s
%s: %s
" $info ($place.people_label | default "Worked with") (. | markdownify) }} {{ end }} {{ $popup := printf "
%s
%sCheck details →" (.name | htmlEscape) $info $k }} {{ $header := printf "

%s

" (.name | htmlEscape) }} {{ with .logo }} {{ $logo := resources.Get (printf "media/icons/brands/%s.svg" .) }} {{ if not $logo }}{{ errorf "places_map: logo not found at assets/media/icons/brands/%s.svg" . }}{{ end }} {{ $header = printf "" $logo.RelPermalink ($place.name | htmlEscape) }} {{ end }} {{ $card := printf "%s%s" $header $info }} {{ with .card_text }}{{ $card = printf "%s
%s
" $card (. | markdownify) }}{{ end }} {{ $points = $points | append (dict "lat" .lat "lng" .lng "popup" $popup "card" $card) }} {{ end }} {{ with $places }}
{{ end }}
